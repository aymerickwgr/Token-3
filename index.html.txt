    let currentVideo = null;


    /* =====================================================
       AFFICHER LES VIDÉOS
       ===================================================== */

    function renderVideos(list) {

      grid.innerHTML = "";

      if (list.length === 0) {
        emptyState.style.display = "block";
        return;
      }

      emptyState.style.display = "none";

      list.forEach(video => {

        const card = document.createElement("article");

        card.className = "video-card";

        card.innerHTML = `
          <div class="thumbnail">

            <img
              src="${video.image}"
              alt="${video.title}"
              loading="lazy"
            >

            <span class="views">👁 ${video.views}</span>

            <span class="duration">
              ${video.duration}
            </span>

          </div>

          <div class="video-info">

            <h3>${video.title}</h3>

            <div class="creator">
              <span>
                ${video.creator}
                <span class="verified">✓</span>
              </span>

              <span>${video.category}</span>
            </div>

          </div>
        `;

        card.addEventListener("click", () => openVideo(video));

        grid.appendChild(card);

      });

    }


    /* =====================================================
       OUVRIR UNE VIDÉO
       ===================================================== */

    function openVideo(video) {

      currentVideo = video;

      document.getElementById("modalTitle").textContent =
        video.title;

      document.getElementById("modalMeta").textContent =
        `${video.creator} · ${video.views} vues · ${video.category}`;

      const player = document.getElementById("modalVideo");

      player.src = video.video;

      document.getElementById("videoModal")
        .classList.add("show");

      document.body.style.overflow = "hidden";

      player.play().catch(() => {});

      updateButtons();

    }


    /* =====================================================
       FERMER
       ===================================================== */

    function closeModal() {

      const modal = document.getElementById("videoModal");
      const player = document.getElementById("modalVideo");

      player.pause();
      player.src = "";

      modal.classList.remove("show");

      document.body.style.overflow = "";

    }


    /* =====================================================
       LIKE
       ===================================================== */

    function toggleLike() {

      if (!currentVideo) return;

      const key = "liked_" + currentVideo.id;

      const liked = localStorage.getItem(key) === "true";

      localStorage.setItem(key, String(!liked));

      updateButtons();

      showToast(
        !liked
          ? "❤️ Vidéo ajoutée à tes likes"
          : "Like retiré"
      );

    }


    /* =====================================================
       SAUVEGARDE
       ===================================================== */

    function toggleSave() {

      if (!currentVideo) return;

      const key = "saved_" + currentVideo.id;

      const saved = localStorage.getItem(key) === "true";

      localStorage.setItem(key, String(!saved));

      updateButtons();

      showToast(
        !saved
          ? "⭐ Vidéo sauvegardée"
          : "Vidéo retirée des sauvegardes"
      );

    }


    /* =====================================================
       METTRE À JOUR LES BOUTONS
       ===================================================== */

    function updateButtons() {

      if (!currentVideo) return;

      const liked =
        localStorage.getItem("liked_" + currentVideo.id) === "true";

      const saved =
        localStorage.getItem("saved_" + currentVideo.id) === "true";

      const likeBtn =
        document.getElementById("likeBtn");

      const saveBtn =
        document.getElementById("saveBtn");

      likeBtn.classList.toggle("liked", liked);
      saveBtn.classList.toggle("saved", saved);

      likeBtn.textContent =
        liked ? "♥ J'aime" : "♡ J'aime";

      saveBtn.textContent =
        saved ? "★ Sauvegardée" : "☆ Sauvegarder";

    }


    /* =====================================================
       PARTAGER
       ===================================================== */

    async function shareVideo() {

      if (!currentVideo) return;

      const shareData = {
        title: currentVideo.title,
        text: `Regarde cette vidéo sur Snap FR : ${currentVideo.title}`,
        url: window.location.href
      };

      if (navigator.share) {

        try {
          await navigator.share(shareData);
        } catch (error) {}

      } else {

        await navigator.clipboard.writeText(
          window.location.href
        );

        showToast("🔗 Lien copié !");

      }

    }


    /* =====================================================
       RECHERCHE
       ===================================================== */

    function searchVideos() {

      const query =
        searchInput.value
          .toLowerCase()
          .trim();

      if (!query) {
        currentVideos = [...videos];
        renderVideos(currentVideos);
        return;
      }

      currentVideos = videos.filter(video => {

        return (
          video.title.toLowerCase().includes(query) ||
          video.creator.toLowerCase().includes(query) ||
          video.category.toLowerCase().includes(query)
        );

      });

      renderVideos(currentVideos);

      scrollToVideos();

    }


    searchInput.addEventListener("input", () => {

      if (searchInput.value.trim() === "") {

        currentVideos = [...videos];

        renderVideos(currentVideos);

      }

    });


    searchInput.addEventListener("keydown", event => {

      if (event.key === "Enter") {
        searchVideos();
      }

    });


    /* =====================================================
       CATÉGORIES
       ===================================================== */

    document.querySelectorAll(".category")
      .forEach(button => {

        button.addEventListener("click", () => {

          const category =
            button.dataset.category;

          document.querySelectorAll(".category")
            .forEach(btn =>
              btn.classList.remove("active")
            );

          button.classList.add("active");

          filterCategory(category);

        });

      });


    function filterCategory(category) {

      if (category === "Toutes") {

        currentVideos = [...videos];

      } else {

        currentVideos =
          videos.filter(video =>
            video.category === category
          );

      }

      renderVideos(currentVideos);

      scrollToVideos();

    }


    function resetFilters() {

      document.querySelectorAll(".category")
        .forEach(btn =>
          btn.classList.remove("active")
        );

      document
        .querySelector('[data-category="Toutes"]')
        .classList.add("active");

      searchInput.value = "";

      currentVideos = [...videos];

      renderVideos(currentVideos);

    }


    /* =====================================================
       UTILITAIRES
       ===================================================== */

    function scrollToVideos() {

      document.getElementById("videosSection")
        .scrollIntoView({
          behavior: "smooth"
        });

    }


    function scrollToTop() {

      window.scrollTo({
        top: 0,
        behavior: "smooth"
      });

    }


    function showToast(message) {

      const toast =
        document.getElementById("toast");

      toast.textContent = message;

      toast.classList.add("show");

      setTimeout(() => {

        toast.classList.remove("show");

      }, 2200);

    }


    /* =====================================================
       FERMETURE DU MODAL
       ===================================================== */

    document
      .getElementById("videoModal")
      .addEventListener("click", event => {

        if (event.target.id === "videoModal") {
          closeModal();
        }

      });


    document.addEventListener("keydown", event => {

      if (event.key === "Escape") {
        closeModal();
      }

    });


    /* =====================================================
       INITIALISATION
       ===================================================== */

    renderVideos(videos);

  </script>
